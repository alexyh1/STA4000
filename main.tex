\begin{abstract}
\begin{spacing}{1.5}
\normalsize
\bigskip
Optimal transport provides a mathematical framework for computing the
most cost-efficient way of transporting one probability distribution of mass
into another. In recent years, it has found widespread applications in computer science, physics, statistics, economics, cell biology and other fields.
Topics include: Monge and Kantorovich formulation, Wasserstein distances,
c-cyclical monotonicity, dual problems, entropic regularization, statistical
convergence, gradient flows, generalized optimal transport problems (Weak
OT, unbalanced OT, Wasserstein barycenters, etc). Computational approaches and numerical algorithms will also be introduced. Weekly discussions involve literature reviews on recent research papers and presenting derivations / code implementations from the textbook(s). While these notes closely follow the main text of ”Optimal Transport for Machine Learners” by Gabriel Peyré, I will try to make it instructional by including my own examples / explanations, data exercises and results from literature reviews. Code implementations are available at \url{https://github.com/alexyh1/STA4000}.
\newline
\newline
Pre/Co-Requisites: Graduate Real Analysis, Graduate Probability.
\newline
Supplemental Readings: Statistical Optimal Transport by Chewi et al, Topics in Optimal Transportation by Villani.

\end{spacing}
\end{abstract}

\section{Monge Formulation}

We will first proceed with some prerequisite knowledge for formulating the Optimal Transport problem.

\subsection{Measures}
\begin{definition}[Probability Simplex]
    The probability simplex of length $n$ is defined by $\Sigma_n = \{a \in \mathbb{R}^n_{\geq 0}: \sum_{i = 1}^n a_i = 1\}$.
    \newline
    Elements of the $\Sigma_n$ are commonly called probability vectors.
\end{definition}

\begin{definition}[Discrete Measure]
A discrete measure with weights $a$ $x_1, ..., x_n \in X$ is defined by $\mu = \sum_{i = 1}^n a_i \delta_{x_i}$ where $\delta_{x_i}$ is the Dirac measure located at $x_i \in \mathbb{R}$.
\newline
$\mu$ is a probability measure if $a \in \Sigma_n$ ($\mu: X \rightarrow [0, \infty]$ such that $\mu(X) = 1$, monotonicity, sub-additivity and $\sigma$-additivity hold).
\end{definition}

We require sufficient regularity conditions for many of the results in the book; unless otherwise mentioned, we will be working with positive Borel measures $\mu \in \mathcal{M}(X)$ on a topological metric space $(X, d)$ where $\mu$ is defined on all Borel sets (Complements, unions, or intersections of open sets in $X$). In general, we will be working on positive probability measures denoted $\mu \in \mathcal{M}(X)$. We usually also assume that the metric space $(X, d)$ is Polish such that it is complete (Every Cauchy sequence converges) and there exists a countable dense set in $X$ (Ex. The general topology on $\mathbb{R}$ is Polish, whereas the topology induced by the discrete metric is not).

\begin{definition}[Support of Measure]
    The support of a Borel measure $\mu$ on a metric space ($X$, d) is given by $\text{supp}(\mu) = \{x \in X: \mu(B_{\varepsilon}(x)) > 0 \ \forall \varepsilon > 0\}$.
\end{definition}

\begin{definition}[Radon Measure]
A Radon measure $\mu$ on a metric space $X$ is a Borel measure such that $\mu(K) < \infty$ for all compact sets $K \subset X$, $\mu(E) = \inf\{\mu(U): U \supset E, U \text{ is open}\}$ for Borel sets $E$ and $\mu(A) = \sup\{\mu(B): B \subset A, B \text{ is compact}\}$.
\end{definition}

We can restrict our attention to Radon measures as every Polish space is a Radon space, such that Borel probability measures in a Polish space (Which we consider) are indeed Radon measures by definition.

\begin{proposition}[Discretization of Lebesgue Integral]
    Suppose $f$ is a measurable function ($\{x: f(x) < t\}$ are Borel sets wrt $\mu$), then $\langle f, \mu \rangle = \displaystyle \int_X f(x)d\mu(x) = \sum_{i = 1}^n a_i f(x_i)$
\end{proposition}

The discretization follows from Riesz-Markov-Kakutani's Theorem such that integration on a finite compact space and measure admits a continuous linear functional on the Banach space of continuous functions and vice-versa.

\begin{definition}[Relative Density]
    If $\mu$ is absolutely continuous to a reference measure $\lambda$ denoted $\mu << \lambda$ ($\lambda(A) = 0 \Rightarrow \mu(A) = 0$) then the relative density is its Radon-Nikodym derivative $\rho_\mu = \frac{d\mu}{d\lambda}$, and for all continuous functions $h$ on $X$, $\displaystyle \int_X h(x)d\mu(x) = \displaystyle \int_X h(x) \rho_\mu(x) d\lambda(x)$.
\end{definition}

\begin{definition}[Total Variation]
    The total variation of a finite signed Radon measure $\mu$ on a compact space $X$ is given by $\lvert \mu \rvert_{TV} = \sup_{f \in \mathcal{C}(X)}\{\langle f, \mu \rangle: \lvert f \rvert _\infty \leq 1\}$
\end{definition}

It follows trivially that total variation of probability measures are always equal to 1.

\begin{definition}[Absolute Value of Measures]
    The measure-theoretic equivalent definition of total variation is given by its absolute value denoted as $\lvert \mu \rvert (A) = \sup_{A = \cup_i B_i} \sum_i \lvert \mu(B_i) \rvert$ ie finite or countable measurable partitions of $A$.
\end{definition}

\begin{example}
    Let $\mu$ be a measure that is absolutely continuous to the Lebesgue measure such that $d\mu = (3x -2) \mathds{1}_{[-1, 1]}dx$. Then the total variation of $\mu$ via 1.7 and 1.8 is given by 
    \[
    \lvert \mu([-1, 1]) \rvert = \lvert \mu([-1, \frac{2}{3}])| + \lvert \mu((\frac{2}{3}, 1]) \rvert,
    \]
    Then the Lebesgue integrals can be easily computed via 1.6 as
    \[
    \lvert \mu \rvert ([-1, 1]) =  \lvert \displaystyle \int_{-1}^{\frac{2}{3}} (3x-2)dx \rvert + \displaystyle \int_{\frac{2}{3}}^1 (3x-2)dx = \frac{25}{6}+\frac{1}{6} = \frac{13}{3}.
    \]
\end{example}

Notice that we can verify that the definitions of total variation are indeed equivalent via the following proposition.

\begin{proposition}[Dual Definition of Total Variation]
    For a finite signed Radon measure $\mu$ on a compact space $X$, we have $\lvert \mu \rvert_{TV} = \lvert \mu \rvert (X)$.
\end{proposition}

\begin{proof}
    To show the forward inequality, it follows that
    \[
    \lvert \displaystyle \int f d\mu \rvert \leq \displaystyle \int \lvert f \rvert \ d \lvert \mu \rvert \leq \lvert f \rvert_\infty \lvert \mu \rvert (X).
    \]
    For the reverse inequality, let $\mu = \mu^+ - \mu^-$ via Jordan decomposition such that $\lvert \mu \rvert = \mu^+ + \mu^-$, then $s = \frac{d\mu}{d\lvert \mu \rvert}$ = -1 or 1 excluding null sets, such that $d \mu = s \ d \lvert \mu \rvert$. By regularity of Radon measures on compact spaces we can find a sequence of continuous functions $f_k \in L^1 (\lvert \mu \rvert)$ (i.e. $\displaystyle \int \lvert f_k \rvert \ d\mu < \infty$) that approximates $s$ with $\lvert f_k \rvert_\infty \leq 1$, therefore
    \[
    \displaystyle \int f_k d\mu \rightarrow \displaystyle \int s d\mu = \lvert \mu \rvert (X) . \qedhere.
    \]   
\end{proof}

\begin{remark}
    For continuous measures $d\mu = \rho_\mu d\lambda$ and $d\nu = \rho_\nu d\lambda$ we obtain $\lvert \mu - \nu \rvert_{TV} = \displaystyle \int_X \lvert \rho_\mu(x) - \rho_\nu(x) \rvert d\lambda(x)$, whereas the discrete formula is given by $\lvert \mu - \nu \rvert_{TV} = \sum_{k} \lvert a_k - b_k \rvert$ when $\mu = \sum_k a_k \delta_{z_k}$ and $\nu = \sum_k b_k \delta_{z_k}$ (Unionize supports and set missing coefficients to 0).
    \newline
    For probability measures, integration cleanly simplifies to $\displaystyle \int_X f(x) d\mu(x) = \mathbb{E}(f(X))$.
\end{remark}


\subsection{Push-Forwards}
\begin{definition}[Push-Forward]
    For a continuous map $T: X \rightarrow Y$, the push-forward measure is $\nu = T_\# \mu \in \mathcal{M}(Y)$ such that $\mu \in \mathcal{M}(X)$ satisfies
    \[
    \forall h \in \mathcal{C}(Y), \displaystyle \int_Y h(y)d\nu(y) = \displaystyle \int_X h(T(x))d\mu(x),
    \]
    Where $T_\#: \mathcal{M}(X) \rightarrow \mathcal{M}(Y)$ is the push-forward map.
    Equivalently, $\nu(B) = \mu(T^{-1}(B))$.
\end{definition}

\begin{remark}
    Discrete measures follow neatly such that for $\mu = \sum_i a_i \delta_{x_i}$, its push-forward measure is given by $\nu = \sum_i a_i \delta_{T(x_i)}$.
\end{remark}

The existence of push-forward map begs the question of whether a function can be "pulled back" into the domain of $T$ / metric space $X$, which is outlined in the definition below. 

\begin{definition}[Pull-Back]
    The pull-back by $T$ denoted $T^\#: \mathcal{C}(Y) \rightarrow \mathcal{C}(X)$ such that for a continuous function $g$, $T^\#g = g \circ T$. From Definition 1.10 we immediately obtain
    \[
    \displaystyle \int_X T^\#g(x)d\mu(x) = \displaystyle \int_Y g(y)d(T_\#\mu)(y).
    \]
\end{definition}

\begin{theorem}[Change of Variables for Relative Densities]
    Let $\mu, \nu$ have relative densities $\rho_\mu, \rho_\nu$ on $\mathbb{R}^d$, if $T$ is a $C^1$ diffeomorphism and $T$ pushes $\mu$ to $\nu$, then
    \[
    \rho_\nu(T(x)) = \rho_\mu(T^{-1}(y))\frac{1}{\det(T'(T^{-1}(y)))}.
    \]
\end{theorem}

\begin{proof}
    Using the Radon-Nikodym formulation of the relative densities (Definition 1.6) and setting $y = T(x)$ along with change of variables we get that
    \[
    \displaystyle \int_Y h(y)\rho_\nu(y)dy = \displaystyle \int_Y h(y) d\nu(y) = \displaystyle \int_X h(T(x))d\mu(x) \]
    \[ = \displaystyle \int_X h(T(x)) \rho_\mu dx = \displaystyle \int_Y h(y) \rho_\mu(T^{-1}(y))\frac{dy}{\det(T'(T^{-1}(y)))}
    \]
    for all $h$ therefore equality of the integrands hold. By bijectivity of $T$ we get $\rho_\mu(x) = \lvert \det(T'(x))\rvert \rho_\nu(T(x))$ completing the proof.
\end{proof}
